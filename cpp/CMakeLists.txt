cmake_minimum_required(VERSION 3.20)
project(yomiyasu_ppdd LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # Backend plug-ins use a C ABI. Core libraries below must share a CRT
  # because libllama passes FILE* to the GGUF reader in ggml-base.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
endif()
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
  set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
  set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
endif()
include(FetchContent)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
FetchContent_Declare(llama
  GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
  GIT_TAG 0a1ef309cdbb9f2024f58a38fb4c42352c5db569
  GIT_PROGRESS TRUE)
FetchContent_MakeAvailable(llama)
if(MSVC)
  foreach(core ggml-base ggml ggml-cpu ggml-cpu-feats llama vendor-hash)
    set_property(TARGET ${core} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  endforeach()
endif()
add_executable(ppdd-worker ppdd_worker.cpp ${llama_SOURCE_DIR}/vendor/hash/sha256/sha256.c)
target_include_directories(ppdd-worker PRIVATE ${llama_SOURCE_DIR}/vendor ${llama_SOURCE_DIR}/vendor/hash)
target_link_libraries(ppdd-worker PRIVATE llama ggml)
if(MSVC)
  set_property(TARGET ppdd-worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(ppdd-worker PRIVATE /utf-8)
endif()
set_target_properties(ppdd-worker PROPERTIES BUILD_RPATH "$ORIGIN;@loader_path" INSTALL_RPATH "$ORIGIN;@loader_path")
install(TARGETS ppdd-worker RUNTIME DESTINATION bin)
